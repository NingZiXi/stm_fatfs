cmake_minimum_required(VERSION 3.22)
project(stm_fatfs VERSION 1.0.1 LANGUAGES C)
include(FetchContent)

set(STM_FATFS_SOURCE_DIR "" CACHE PATH "FatFs source directory containing ff.c, ff.h and ffconf.h")
set(STM_FATFS_FETCH_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../fatfs" CACHE PATH "FatFs source checkout/download directory")
option(STM_FATFS_FETCH "Fetch/vendor FatFs when no target or source is supplied" ON)
if(TARGET fatfs)
    set(_fatfs_target fatfs)
else()
    if(STM_FATFS_SOURCE_DIR)
        set(_fatfs_source "${STM_FATFS_SOURCE_DIR}")
    elseif(EXISTS "${STM_FATFS_FETCH_DIR}/ff.c")
        set(_fatfs_source "${STM_FATFS_FETCH_DIR}")
    elseif(STM_FATFS_FETCH)
        message(FATAL_ERROR "FatFs source is absent. Provide STM_FATFS_SOURCE_DIR or a pinned STM_FATFS_FETCH_DIR; automatic download needs an explicit trusted source.")
    else()
        message(FATAL_ERROR "FatFs source is absent and STM_FATFS_FETCH is OFF")
    endif()
    foreach(_file ff.c ff.h ffconf.h)
        if(NOT EXISTS "${_fatfs_source}/${_file}")
            message(FATAL_ERROR "Missing FatFs file: ${_fatfs_source}/${_file}")
        endif()
    endforeach()
    add_library(fatfs STATIC "${_fatfs_source}/ff.c")
    if(EXISTS "${_fatfs_source}/ffunicode.c")
        target_sources(fatfs PRIVATE "${_fatfs_source}/ffunicode.c")
    endif()
    option(STM_FATFS_USE_PORTABLE_SYSTEM "Provide malloc/time hooks for the bundled FatFs" ON)
    if(STM_FATFS_USE_PORTABLE_SYSTEM)
        target_sources(fatfs PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/ffsystem_portable.c")
    elseif(EXISTS "${_fatfs_source}/ffsystem.c")
        target_sources(fatfs PRIVATE "${_fatfs_source}/ffsystem.c")
    endif()
    target_include_directories(fatfs PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include" "${_fatfs_source}")
    target_compile_features(fatfs PUBLIC c_std_11)
    set(_fatfs_target fatfs)
endif()

add_library(stm_fatfs STATIC src/stm_fatfs.c)
target_include_directories(stm_fatfs PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(stm_fatfs PUBLIC c_std_11)
target_link_libraries(stm_fatfs PUBLIC ${_fatfs_target})

if(NOT TARGET stm_common)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../stm_common/CMakeLists.txt")
        add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../stm_common" "${CMAKE_CURRENT_BINARY_DIR}/stm_common")
    else()
        option(STM_COMMON_FETCH "Download stm_common when no local target or sibling exists" ON)
        set(STM_COMMON_GIT_REPOSITORY "https://github.com/NingZiXi/stm_common.git"
            CACHE STRING "stm_common Git repository (GitHub or Gitee mirror)")
        if(NOT STM_COMMON_FETCH)
            message(FATAL_ERROR "stm_common is missing. Provide its target or sibling directory, or enable STM_COMMON_FETCH.")
        endif()
        FetchContent_Declare(stm_common
            GIT_REPOSITORY "${STM_COMMON_GIT_REPOSITORY}"
            GIT_TAG ce3d186dde2d374a8e9c7b9068a7b88f97d57dc1)
        FetchContent_MakeAvailable(stm_common)
    endif()
endif()
target_link_libraries(stm_fatfs PUBLIC stm_common)

option(STM_FATFS_WITH_SD "Build the optional stm_fatfs_sd bridge when stm_sd is available" ON)
if(STM_FATFS_WITH_SD)
    if(NOT TARGET stm_sd AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../stm_sd/CMakeLists.txt")
        add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../stm_sd" "${CMAKE_CURRENT_BINARY_DIR}/stm_sd")
    endif()
    if(TARGET stm_sd)
        add_library(stm_fatfs_sd STATIC adapters/sd/stm_fatfs_sd.c)
        target_include_directories(stm_fatfs_sd PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/adapters/sd)
        target_link_libraries(stm_fatfs_sd PUBLIC stm_fatfs stm_sd)
    else()
        message(STATUS "stm_fatfs_sd skipped: provide stm_sd or set STM_FATFS_WITH_SD=OFF")
    endif()
endif()
